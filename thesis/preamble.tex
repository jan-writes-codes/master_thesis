% preamble.tex — shared packages, math operators, and notation macros.

% --- Encoding & fonts -------------------------------------------------------
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[english]{babel}

% --- Math -------------------------------------------------------------------
\usepackage{amsmath, amssymb, amsthm, bm}
\usepackage{mathtools}

% --- Tables & figures -------------------------------------------------------
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{multirow}

\usepackage{caption}
\usepackage{subcaption}
\usepackage{float}

\graphicspath{{figures/}{tables/}}

% JFE-style table headings: "Table N" and a bold title, each on its own line,
% set above the table; notes go below via \tabnotes.
\captionsetup[table]{position=above, labelfont=bf, textfont=bf,
  labelsep=newline, justification=raggedright, singlelinecheck=false,
  skip=0.6em}
\newcommand{\tabnotes}[1]{\par\vspace{0.6em}%
  \begin{minipage}{\linewidth}%
    \footnotesize\singlespacing\noindent #1\par
  \end{minipage}}
\newcommand{\panel}[2]{\multicolumn{#1}{@{}l}{\textit{#2}}}

% --- Layout -----------------------------------------------------------------
\usepackage[a4paper, margin=2.5cm]{geometry}
\usepackage{setspace}
\onehalfspacing
\usepackage{microtype}
\usepackage{enumitem}

% --- References -------------------------------------------------------------
% Author-year citations via natbib with the chicago bibliography style, the
% referencing convention of the finance academic literature (per the
% supervisor's thesis guidelines). cleveref stays loaded last.
\usepackage[round, authoryear]{natbib}
\usepackage{hyperref}
\hypersetup{colorlinks=true, linkcolor=black, citecolor=black, urlcolor=blue}
\usepackage[capitalise, noabbrev]{cleveref}
