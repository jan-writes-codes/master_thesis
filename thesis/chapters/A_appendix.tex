\section{Additional Figures}
\label{app:figures}
This section collects supporting figures referenced in the main text.

\begin{figure}[htbp]\centering
  \includegraphics[width=0.85\linewidth]{figures/mr_f1_r2_phase1.pdf}
  \caption[In-sample $R^{2}$ of the local regression by country]{In-sample $R^{2}$ of the local predictive regression
    \eqref{eq:h-local}, $\rxbar_{i,t+12}\sim\CF_{i,t}$, by country.}
  \label{fig:mr-r2-phase1}
  \fignotes{Bars report the in-sample $R^{2}$ of the country-level Phase~I regression
  \eqref{eq:h-local} of the duration-standardised, maturity-averaged excess return
  \eqref{eq:rxbar} on the local cycle factor. Because the factor is the fitted
  value of \eqref{eq:cf}, this $R^{2}$ equals the fit of the underlying
  two-predictor forecasting regression reported in \Cref{tab:mr-phase1}. Countries
  are ordered by $R^{2}$ rather than alphabetically.}
\end{figure}

\begin{figure}[htbp]\centering
  \includegraphics[width=0.8\linewidth]{figures/s1_yield_curve_avg.pdf}
  \caption{Average zero-coupon yield curve by country over the sample.}
  \label{fig:yield-curve-avg}
  \fignotes{Each line is a country's time-series average zero-coupon yield at maturities of
  $1,2,4,5,9$ and $10$ years, computed over that country's own sample because the
  panel is unbalanced. Yields are end-of-month Bloomberg fitted zero-coupon
  government curves, continuously compounded and in percent per annum.}
\end{figure}

    between the local cycle factor and the local forward factor.}

\begin{figure}[htbp]\centering
  \includegraphics[width=0.88\linewidth]{figures/rob_f3_core_vs_reg.pdf}
  \caption[In-sample $R^{2}$: core versus headline CPI]{In-sample $R^{2}$ of the local predictive regression
    $\rxbar_{i,t+12}\sim\CF_{i,t}$ by country, with the cycle factor built on core
    (ex food and energy) versus regular all-items CPI.}
  \label{fig:rob-core-headline}
  \fignotes{Bars compare the in-sample $R^{2}$ of \eqref{eq:h-local} when trend inflation
  \eqref{eq:trendinf} is built on core CPI, excluding food and energy, against
  all-items CPI. Only the inflation series changes. The window $M=120$ and decay
  $v=0.987$, the yield--cycle decomposition and the factor construction are held
  at their baseline values. Core CPI is the baseline used throughout the thesis.}
\end{figure}

\begin{figure}[htbp]\centering
  \includegraphics[width=0.9\linewidth]{figures/mr_f2_r2_ladder.pdf}
  \caption[Phase~II in-sample $R^{2}$ ladder by country]{Phase~II in-sample $R^{2}$ ladder by country: the local factor alone
    \eqref{eq:h-local}, the global factor alone \eqref{eq:h-global}, and the
    joint horse race \eqref{eq:h-horse}.}
  \label{fig:mr-r2-ladder}
  \fignotes{For each country the three bars report the in-sample $R^{2}$ of the local factor
  alone \eqref{eq:h-local}, the global factor alone \eqref{eq:h-global}, and the
  joint regression \eqref{eq:h-horse} in which the local factor enters
  orthogonalised. Under full integration the third bar should not exceed the
  second, since the orthogonalised local factor then adds nothing.}
\end{figure}

\begin{figure}[htbp]\centering
  \includegraphics[width=0.9\linewidth]{figures/s1_yield_ts.pdf}
  \caption[Zero-coupon yields across the G10 by country and maturity]{Zero-coupon yields across the G10 by country and maturity,
    1989--2026. The secular decline in nominal yields and the post-2021 reversal
    are common across countries and across the curve.}
  \label{fig:yield-ts}
  \fignotes{One panel per country, with a line for each of the six maturities
  $\{1,2,4,5,9,10\}$ years. Yields are end-of-month Bloomberg fitted zero-coupon
  government curves, continuously compounded and in percent per annum. Vertical
  scales are free across panels, so levels should be compared within a panel
  rather than between panels. The series run over the full available history of
  the raw curves, which extends beyond the April 2024 end of the estimation
  sample.}
\end{figure}

  \caption[Recursive versus full-sample global cycle factor]{Left: the full-sample global cycle factor $\GCF_{t}$ against the
    fully-recursive $\GCF^{\mathrm{oos}}_{t}$; convergence after the trend-inflation
    burn-in signals a stable real-time estimator. Right: per-country correlation
    between the full-sample and recursive \emph{local} cycle factors, where a
    value closer to one means that the full-sample look-ahead added little.}
  \label{fig:s8-gcf-conv}
  \fignotes{The left panel plots the full-sample global cycle factor $\GCF_{t}$
  \eqref{eq:gcf} against its fully-recursive counterpart
  $\GCF^{\mathrm{oos}}_{t}$, which is rebuilt at each date from data available
  then. The two are expected to differ early in the sample, while the
  trend-inflation average \eqref{eq:trendinf} is still filling its window. The
  right panel reports the per-country correlation between the full-sample and
  recursive local factors, where a value near one means the full-sample estimate
  added little look-ahead information. In the right panel countries are ordered by
  correlation rather than alphabetically.}
\end{figure}
