\section{Yield Decomposition and Interest-Rate Cycles}
\label{sec:fw-decomp}
% Eq (1)-(2): y^{(n)}_{i,t} = alpha_{i,n} + beta_{i,n} pi^e_{i,t} + c^{(n)}_{i,t}.
% Eq (3): trend inflation as an EWMA of YoY inflation (M=120, decay v).
\begin{equation}
  y^{(n)}_{i,t} = \alpha_{i,n} + \beta_{i,n}\,\trendinf_{i,t} + \cyc^{(n)}_{i,t},
  \qquad \cyc^{(n)}_{i,t} = \varepsilon^{(n)}_{i,t}.
  \label{eq:decomp}
\end{equation}
\begin{equation}
  \trendinf_{i,t} = \frac{1-v}{1-v^{M}}\sum_{j=0}^{M-1} v^{j}\,\pi_{i,t-j},
  \qquad M = 120.
  \label{eq:trendinf}
\end{equation}

\section{Local Cycle Factor}
\label{sec:fw-localcf}
% Eq (4)-(6): rx_{i,t+12} ~ c^{(1)}_{i,t} + cbar_{i,t}; CF = fitted value.
\begin{align}
  \rxbar_{i,t+12} &= \gamma_{0} + \gamma_{i,1}\,\cyc^{(1)}_{i,t}
                     + \gamma_{i,2}\,\cycbar_{i,t} + \varepsilon_{i,t+12}, \label{eq:cf-reg}\\
  \cycbar_{i,t}   &= \tfrac{1}{|N|}\textstyle\sum_{n\in N}\cyc^{(n)}_{i,t}, \label{eq:cbar}\\
  \CF_{i,t}       &= \hat\gamma_{i,1}\,\cyc^{(1)}_{i,t} + \hat\gamma_{i,2}\,\cycbar_{i,t}. \label{eq:cf}
\end{align}

\section{Global Cycle Factor and GDP Weights}
\label{sec:fw-gcf}
% Eq (7)-(8): GCF as GDP-weighted average of local CF.
\begin{equation}
  \GCF_{t} = \sum_{i\in G10} w_{i,t}\,\CF_{i,t},
  \qquad w_{i,t} = \frac{\mathrm{GDP}_{i,t}}{\sum_{j\in G10}\mathrm{GDP}_{j,t}}.
  \label{eq:gcf}
\end{equation}

\section{FX-Adjusted Global Cycle Factor}
\label{sec:fw-fxgcf}
% Eq (15)-(17): GDP-weighted average USD return; project on GCF; FXGCF = fitted.
\begin{equation}
  \overline{\rx}^{\,\mathrm{USD}}_{t+12} = \delta_0 + \delta_1\,\GCF_t + \varepsilon_{t+12},

\end{equation}
