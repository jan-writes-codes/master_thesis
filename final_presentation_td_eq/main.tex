\section{Motivation \& Research Question}

% Motivation slide (mirrors the proposal's opener): build from the EH baseline
% and the predictability literature to the international gap that the RQ fills.
\begin{frame}{Motivation}
  \textbf{The expectation hypothesis (EH):} a long yield is just the average
  expected future short rate, so bond \emph{excess} returns should be
  unpredictable.
  \begin{equation*}
    y^{(n)}_{t} \;\approx\; \frac{1}{n}\sum_{i=0}^{n-1}
      \E_{t}\!\left[y^{(1)}_{t+i}\right]
  \end{equation*}
  Yet the \alert{evidence finds them predictable}, consistent with time-varying risk premia:
  \begin{enumerate}
    \item \citet{cochrane2005}: a tent-shaped forward-rate factor (CP) strongly
      predicts US Treasury returns, but offers no economic mechanism.
    \item \citet{cieslak2015}: splitting yields into \emph{trend inflation} and
      \emph{cycles}, the cycle factor predicts returns through a macro-anchored
      mechanism.
    \item \citet{dahlquist2013}: a \emph{global} CP factor predicts international
      returns better than local factors alone, implying integrated bond markets.
  \end{enumerate}
  \vfill
  \textbf{The gap:} the macro-anchored cycle factor is established for the US.
  Whether it extends across countries and aggregates into a \emph{unified global
  cycle factor} is open \citep[cf.][]{bauer2020,zhang2021}.
\end{frame}

% Research question (stated immediately after the one motivation slide)
\begin{frame}{Research Question}
  \begin{block}{Main Question}
    Does the \citet{cieslak2015} cycle factor (CF) apply to international
    bonds?\\
    {\small\color{gray} If their propositions hold, the mechanism found in the
    US market should be fundamental to bond markets in other developed
    countries.}
  \end{block}
  \textbf{Subquestions}
  \begin{enumerate}
    \item What is the role of the local CF in comparison to a GDP-weighted
      global CF ($\GCF$)?
    \item Does a global CF subsume the predictive power of a local CF?
    \item Does currency risk break the predictability for a USD investor, and
      does an FX-adjusted global factor ($\FXGCF$) restore it?
  \end{enumerate}
\end{frame}

\begin{frame}{Recap: The Cycle factor framework}
  Yields decompose into slow-moving \emph{trend inflation} and transitory
  \emph{cycles} \citep{cieslak2015}:
  \begin{equation*}
    y^{(n)}_{i,t} \;=\; \alpha_{i,n} \;+\; \beta_{i,n}\,\pi^{e}_{i,t}
      \;+\; \underbrace{c^{(n)}_{i,t}}_{\text{cycle}}
  \end{equation*}
  The cycles forecast one-year-ahead, duration-standardised,
  maturity-averaged excess returns $\rxbar_{i,t+12}$.
  \vfill
  \textbf{Factor hierarchy}
  \begin{enumerate}
    \item \textbf{Local} $\CF_{i,t}$: fitted return forecast from
      $c^{(1)}_{i,t}$ and the average cycle $\bar c_{i,t}$
    \item \textbf{Global} $\GCF_{t} = \sum_{i} w_{i,t}\,\CF_{i,t}$:
      GDP-weighted shared risk \citep{dahlquist2013}
    \item \textbf{FX-adjusted global} $\FXGCF_{t}$: built on the GDP-weighted
      average \emph{USD} return for an unhedged US investor
  \end{enumerate}
\end{frame}

\section{Research Design \& Data}

% MTS requirement: one slide outlining the research design
\begin{frame}{Research Design}
  \textbf{Sample:} G10 markets, monthly zero-coupon yields, core CPI, FX spot
  rates, and nominal GDP over 1994--2026.
  \vfill
  \textbf{Three empirical phases} (in-sample, Newey--West HAC inference)
  \begin{itemize}
    \item[\textbf{I}] Local predictability:
      $\rxbar_{i,t+12}=\alpha_i+\beta_i\,\CF_{i,t}+\varepsilon$ per country
    \item[\textbf{II}] Integration horse race:
      $\rxbar_{i,t+12}=\alpha_i+\beta_i\,\CF^{\perp}_{i,t}
        +\gamma_i\,\GCF_{t}+\varepsilon$
    \item[\textbf{III}] USD investor:
      $\rxbar^{\,\mathrm{USD}}_{i,t+12}$ on $\GCF_{t}$ vs.\ $\FXGCF_{t}$
  \end{itemize}
  \vfill
  \textbf{Validation \& stress tests}
  \begin{itemize}
    \item Replication of \citet{cieslak2015} and \citet{dahlquist2013} on US /
      international data before any extension
    \item Fully-recursive out-of-sample evaluation
      \citep[Campbell--Thompson $R^{2}_{\mathrm{oos}}$,][]{campbellthompson2008}
    \item Robustness: crisis subsamples, rolling windows, CF vs. CP factor,
      core vs. headline inflation
  \end{itemize}
\end{frame}

\begin{frame}{Data \& Validation}
  \begin{itemize}
    \item \textbf{Universe:} G10 (BE, CA, FR, DE, IT, JP, NL, SE, CH, UK, US)
    \item \textbf{Yields:} zero-coupon curves, maturities 1, 2, 4, 5, 9, 10
      years (Bloomberg)
    \item \textbf{Inflation:} core CPI (FRED)
    \item \textbf{FX:}
      spot against USD (Bloomberg)
    \item \textbf{GDP:} nominal (LSEG)
    \item \textbf{Frequency / span:} monthly, 1994--2026. Returns are
      one-year-ahead, duration-standardised, and maturity-averaged
  \end{itemize}
  \vfill
  \begin{block}{Replication as validation}
    Before testing the factor internationally, we reproduce the published US results: Similar factor correlations, $R^2$ and conclusions for the local US cycle factor.
  \end{block}
\end{frame}

\section{Results}

\begin{frame}{Phase I: Local Cycle factor works across the G10}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Sign pattern of \citet{cieslak2015} ($\gamma_{1}<0$ on the
          1y cycle, $\gamma_{2}>0$ on the average cycle) in \textbf{all 11
          markets}
        \item Joint Wald rejects no-predictability at 1\% in 9 countries,
          at 5\% in 10 (exception: Switzerland, $p=0.20$)
        \item Pooled panel: $t=-10.7$ and $t=15.5$
        \item In-sample $R^{2}$: 7\% (CH) to 34\% (NL), \textbf{average
          25\%}, with the US at 33\%, matching the original study
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \includegraphics[width=\linewidth]{figures/mr_f1_r2_phase1.pdf}
      {\scriptsize In-sample $R^{2}$ of $\rxbar_{i,t+12}\sim\CF_{i,t}$ by
        country.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phase II: One global factor summarises eleven local ones}
  \centering
  \includegraphics[width=0.62\linewidth]{figures/s5_gcf.pdf}
  \begin{itemize}
    \item $\GCF_{t}$ (GDP-weighted) tracks the centre of mass of the eleven
      local factors
    \item On its own, the global factor explains as much as the eleven local
      ones: both average an $R^{2}$ of \textbf{25\%}
  \end{itemize}
\end{frame}

\begin{frame}{Phase II: GCF largely subsumes the local}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/mr_f2_hr_tstats.pdf}
      {\scriptsize HAC $t$-statistics in the horse race
        $\rxbar\sim\CF^{\perp}+\GCF$. Dashed lines $\pm1.96$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $\GCF$ significant in \textbf{every} market
          ($t$ between 2.1 and 4.5)
        \item Orthogonalised local factor survives in only
          \textbf{3 of 11} markets: Italy ($t=3.7$), Netherlands ($t=3.3$),
          Belgium ($t=2.4$)
        \item Where subsumed, adding $\CF^{\perp}$ lifts $R^{2}$ only
          trivially
        \item Integration is strong but not complete \citep{dahlquist2013}:
          the three holdouts are all euro-area, where country-specific
          sovereign-spread risk likely keeps a local signal alive
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phase III: USD investor}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/mr_f3_usd_r2.pdf}
      {\scriptsize USD-investor in-sample $R^{2}$: $\GCF$ vs.\ $\FXGCF$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Average $R^{2}$ collapses from \textbf{25\% (local currency) to
          8\% (USD returns)}
        \item Significance lost across the euro area, Switzerland, and Japan, but
          survives in US (33\%), Canada ($t=2.9$), and marginally SE and UK
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phase III: The FX adjustment helps, somewhat less collinear}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/mr_f4_gcf_fxgcf.pdf}
      {\scriptsize $\GCF_{t}$ and $\FXGCF_{t}$: correlation $0.94$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $\FXGCF$ raises the USD-investor $R^{2}$ in \textbf{9 of 11}
          markets and the $t$-statistic in \textbf{9 of 11}
        \item Average $R^{2}$ \textbf{$8\%\to11\%$}, uniform in direction
        \item $\FXGCF$ and $\GCF$ correlate \textbf{0.94} -- below the GDP
          top-down's $0.99$, but still well above the $\approx0.50$ of
          \citet{dahlquist2013}: $1/n$ weighting alone does not decouple it
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{OOS regressions}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/mr_f5_oos_r2.pdf}
      {\scriptsize Campbell--Thompson $R^{2}_{\mathrm{oos}}$ vs.\ the
        recursive prevailing mean, by country.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Local CF:} pooled $R^{2}_{\mathrm{oos}}=-0.08$ and
          beats the mean in only 2 of 11 markets
        \item \textbf{Global $\GCF$:} pooled $\mathbf{+0.05}$, positive in
          \textbf{10 of 11} markets, clearest evidence of integration
        \item \textbf{USD investor:} $\GCF$ at $-0.07$; the $1/n$ $\FXGCF$ turns
          it positive ($+0.027$, 8 of 11 markets) -- the strongest OOS of the
          constructions
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Economic value: Timing a global bond portfolio}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\linewidth]{figures/strat_f1_cumret.pdf}
      {\scriptsize Out-of-sample cumulative returns, currency-hedged $\GCF$
        investor, 2005--2024.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Investable product:} one GDP-weighted, 10y G10 government-bond
      basket, timed by a mean--variance rule on the recursive factor.
      \vfill
      \textbf{$\GCF$: currency-hedged}\\
      {\scriptsize (local-currency returns, FX hedged with forwards)}
      \begin{itemize}
        \item $R^{2}_{\mathrm{oos}}=0.125$, Sharpe \textbf{0.31$\to$0.38}
        \item CER ($\gamma=5$): \textbf{+0.51\%/yr} vs.\ buy-and-hold
      \end{itemize}
      \textbf{$\FXGCF$: same basket, unhedged in USD}
      \begin{itemize}
        \item $R^{2}_{\mathrm{oos}}=0.13$, Sharpe \textbf{0.20$\to$0.26}
        \item Best dollar-investor timing of the constructions, still capped by
          one-year FX risk below the hedged investor
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Robustness}
  \textbf{$\GCF$ survives tests}
  \begin{itemize}
    \item The GDP-weighted global cycle factor: in-sample and out-of-sample,
      across subsamples, with core inflation, and
      against the Cochrane--Piazzesi forward-rate factor
  \end{itemize}
  \vfill
  \textbf{Fragile findings}
  \begin{itemize}
    \item Local-factor predictability does not survive real-time evaluation
    \item USD-investor predictability is far weaker (in-sample $R^{2}$ falls
      $25\%\to8\%$) and is negative out of sample for the unadjusted factor
    \item The $1/n$ $\FXGCF$ edge is fairly robust: positive out of sample under
      the expanding 5y/7y and rolling-15y windows, only mildly negative under the
      rolling-10y scheme
  \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{block}{Does the \citet{cieslak2015} cycle factor apply to
      international bonds? Yes.}
    Correct sign structure in all 11 G10 markets, average in-sample
    $R^{2}=25\%$, at every maturity.
  \end{block}
  \begin{enumerate}
    \item \textbf{Local vs.\ global:} one GDP-weighted global factor matches
      eleven national factors and is significant everywhere.
    \item \textbf{Subsumption:} the global factor drives out the local one in
      8 of 11 markets, and predictability is overwhelmingly global. Only
      the global factor predicts \emph{out of sample}.
    \item \textbf{USD investor:} currency risk erodes $\sim$2/3 of the
      predictive power. The $1/n$ top-down $\FXGCF$ restores it ($8\%\to11\%$,
      and is the strongest out of sample at $+0.027$), but stays fairly
      collinear with $\GCF$ (corr \textbf{0.94}).
  \end{enumerate}
  \vfill
  \textbf{Contribution:} a single global cycle factor for the G10, a new
  FX-adjusted version of it, and real-time evidence (including the timing
  strategy) that bond-return predictability is mostly global and rooted in
  inflation trends.
\end{frame}
